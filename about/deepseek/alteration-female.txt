// Global variables to track current selection
let currentAlteration = null;
let currentLevel = null;

// Initialize when DOM is loaded
document.addEventListener('DOMContentLoaded', function() {
    initializeEventListeners();
});

function initializeEventListeners() {
    // Get all alteration select elements
    const alterationSelects = [
        'alteration-bodice-Select',
        'alteration-sleeve-Select', 
        'alteration-skirt-Select',
        'alteration-bustle-Select',
        'alteration-veil-Select',
        'alteration-others-Select'
    ];

    // Add event listeners to all alteration selects
    alterationSelects.forEach(selectId => {
        const select = document.getElementById(selectId);
        if (select) {
            select.addEventListener('change', handleAlterationChange);
        }
    });

    // Add event listener to level select
    const levelSelect = document.getElementById('alterationLevel-diff');
    if (levelSelect) {
        levelSelect.addEventListener('change', handleLevelChange);
    }
}

function handleAlterationChange(event) {
    const selectedValue = event.target.value;
    
    // Reset other selects if one is chosen (only one alteration allowed)
    if (selectedValue) {
        resetOtherSelects(event.target.id);
        currentAlteration = selectedValue;
        updateDisplay();
    } else {
        currentAlteration = null;
        clearDisplay();
    }
}

function handleLevelChange(event) {
    currentLevel = event.target.value;
    updateDisplay();
}

function resetOtherSelects(currentSelectId) {
    const allSelects = [
        'alteration-bodice-Select',
        'alteration-sleeve-Select', 
        'alteration-skirt-Select',
        'alteration-bustle-Select',
        'alteration-veil-Select',
        'alteration-others-Select'
    ];

    allSelects.forEach(selectId => {
        if (selectId !== currentSelectId) {
            const select = document.getElementById(selectId);
            if (select) {
                select.value = '';
            }
        }
    });
}

function updateDisplay() {
    if (!currentAlteration || !currentLevel) return;

    const alteration = alterationMap[currentAlteration];
    if (!alteration) return;

    // Update description
    const descriptionElement = document.getElementById('alteration-description');
    if (descriptionElement) {
        descriptionElement.textContent = alteration.detail;
    }

    // Update alteration type in order summary
    const typeOrderElement = document.getElementById('alteration-type-order');
    if (typeOrderElement) {
        typeOrderElement.textContent = `Alteration: ${currentAlteration}`;
    }

    // Update level in order summary
    const levelOrderElement = document.getElementById('alteration-level-order');
        if (levelOrderElement) {
        const price = alteration[currentLevel];
        if (price === 0) {
            levelOrderElement.textContent = '';
        } else {
            levelOrderElement.textContent = `Level: ${currentLevel}`;
        }
    }

    // Update price calculation
    const priceElement = document.getElementById('priceCalculation');
    if (priceElement) {
        const price = alteration[currentLevel];
        priceElement.textContent = price === 0 ? 'n.a' : `$${price}`;
    }
}

function clearDisplay() {
    const elementsToClear = [
        'alteration-description',
        'alteration-type-order', 
        'alteration-level-order',
        'priceCalculation'
    ];

    elementsToClear.forEach(id => {
        const element = document.getElementById(id);
        if (element) {
            element.textContent = id === 'priceCalculation' ? '' : `${id.replace('-', ' ')} on this portion`;
        }
    });
}

// Form button handlers
function handleAdd() {
    if (!currentAlteration || !currentLevel) {
        alert('Please select both an alteration type and difficulty level');
        return;
    }
    
    const alteration = alterationMap[currentAlteration];
    const price = alteration[currentLevel];
    
    if (price === 0) {
        alert('This service is not available at the selected difficulty level');
        return;
    }
    
    // Here you would typically add to cart or process the order
    alert(`Added: ${currentAlteration} (${currentLevel}) - $${price}`);
}

function handleClear() {
    // Reset all selects
    const allSelects = [
        'alteration-bodice-Select',
        'alteration-sleeve-Select', 
        'alteration-skirt-Select',
        'alteration-bustle-Select',
        'alteration-veil-Select',
        'alteration-others-Select',
        'alterationLevel-diff'
    ];

    allSelects.forEach(selectId => {
        const select = document.getElementById(selectId);
        if (select) {
            select.value = '';
        }
    });

    // Reset global variables
    currentAlteration = null;
    currentLevel = null;
    
    // Clear display
    clearDisplay();
}

function cancelForm() {
    // Simple back functionality
    window.history.back();
}
